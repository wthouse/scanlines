{{- /* JSON-LD structured data. BlogPosting for posts (the configured contentSection), WebPage for other regular pages (About, etc.), WebSite on the home page, plus a Person node when the home page is in profile mode. Built as dicts and piped through `jsonify` so all values are safely escaped. */ -}} {{- $contentSection := site.Params.scanlines.contentSection | default "posts" -}} {{- if and .IsPage (eq .Section $contentSection) -}} {{- $author := .Params.author | default site.Params.author | default site.Title -}} {{- $desc := .Description | default (.Summary | plainify) | chomp -}} {{- $schema := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "url" .Permalink "mainEntityOfPage" .Permalink "author" (dict "@type" "Person" "name" $author) "publisher" (dict "@type" "Organization" "name" site.Title) -}} {{- with $desc }}{{ $schema = merge $schema (dict "description" .) }}{{ end -}} {{- if not .PublishDate.IsZero }}{{ $schema = merge $schema (dict "datePublished" (.PublishDate.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- if not .Lastmod.IsZero }}{{ $schema = merge $schema (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- with (.Params.image | default site.Params.scanlines.ogImage) }}{{ $schema = merge $schema (dict "image" (. | absURL)) }}{{ end -}} {{- with .Params.tags }}{{ $schema = merge $schema (dict "keywords" (delimit . ", ")) }}{{ end -}} {{- else if .IsPage -}} {{- $desc := .Description | default (.Summary | plainify) | chomp -}} {{- $schema := dict "@context" "https://schema.org" "@type" "WebPage" "name" .Title "url" .Permalink "isPartOf" (dict "@type" "WebSite" "name" site.Title "url" site.BaseURL) -}} {{- with $desc }}{{ $schema = merge $schema (dict "description" .) }}{{ end -}} {{- if not .Lastmod.IsZero }}{{ $schema = merge $schema (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- else if .IsHome -}} {{- $website := dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "url" site.BaseURL -}} {{- with site.Params.description }}{{ $website = merge $website (dict "description" .) }}{{ end -}} {{- if eq (site.Params.scanlines.homepage | default "blog") "profile" -}} {{- $profile := site.Params.scanlines.profile -}} {{- $person := dict "@context" "https://schema.org" "@type" "Person" "name" ($profile.name | default site.Title) "url" site.BaseURL -}} {{- with $profile.subtitle }}{{ $person = merge $person (dict "jobTitle" .) }}{{ end -}} {{- /* markdownify first so emoji shortcodes and markdown resolve the same way they do in the visible bio (profile-header.html) — plainify alone left raw ":robot:" / "**bold**" in the structured data. */ -}} {{- with $profile.bio }}{{ $person = merge $person (dict "description" (. | markdownify | plainify | chomp)) }}{{ end -}} {{- with $profile.avatar }}{{ $person = merge $person (dict "image" (. | absURL)) }}{{ end -}} {{- end -}} {{- end -}}
